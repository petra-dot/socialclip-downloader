# Download History Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Record every completed download from the Download tab, the Queue, and the CLI into a persistent `history.json`, and add a fourth tab to browse, play, reveal, delete, or forget those downloads.

**Architecture:** `core/history.py` (Qt-free) holds an entry model, pure search/filter/sort helpers, and a `HistoryStore` that mirrors `QueueStore` (atomic write, tolerant load). Recording happens once, inside `core.download.download_one`, so all three callers record with no duplicated logic. A fourth Qt tab reads the store and derives each row's status by `stat`-ing the file on load, so status can never go stale.

**Tech Stack:** Python 3.8+, PyQt5, pytest. No new dependency.

**Spec:** `docs/specs/2026-10-18-history-design.md`

## Global Constraints

- `core/` is **Qt-free** by rule (enforced by `tests/test_core_isolation.py`). No PyQt5 import in any `core/` module.
- **Never commit a real social-media URL.** Tests, docs, and commits use synthetic placeholders only (`https://a.com/1`, `https://youtu.be/x`).
- Conventional Commits (`feat`/`fix`/`chore`/`docs`). Never commit to `main` on shared work without the user asking. **Never push without asking.**
- The download-result contract (`core.manifest.DownloadResult`, `SCHEMA_VERSION = "1.0"`) is **unchanged** by this work.
- An unknown key in a stored JSON file **never drops an entry** (the `Queue._coerce_jobs` rule).
- Recording a history line **can never fail a download**.
- Every destructive UI action confirms first, and only ever touches a path recorded in the history.
- Lint gate: `python -m flake8 . --count --select=E9,F63,F7,F82 --show-source --statistics` must print `0`.
- Test gate: `python -m pytest -q` must stay green.

# core/history.py
"""A persistent log of completed downloads.

Qt-free: the GUI and the CLI both use it. Recording happens in
core.download.download_one, so every caller records without duplicating logic.
"""

import json
import os
import uuid
from dataclasses import asdict, dataclass, field
from datetime import datetime, timezone
from typing import List, Optional

from core.paths import history_path

STORE_VERSION = 1

STATUS_OK = "ok"
STATUS_MISSING = "missing"
STATUS_DELETED = "deleted"


@dataclass
class HistoryEntry:
    id: str = field(default_factory=lambda: uuid.uuid4().hex)
    title: str = ""
    url: str = ""
    platform: str = ""
    path: str = ""
    bytes: int = 0
    extension: str = ""
    height: int = 0
    duration_seconds: int = 0
    source: str = ""
    created_at: str = field(
        default_factory=lambda: datetime.now(timezone.utc).isoformat()
    )
    removed_at: str = ""


def entry_from_result(result, source: str = "") -> HistoryEntry:
    """Map a core.manifest.DownloadResult to a HistoryEntry.

    The result already carries the title, platform, path, extension, height,
    bytes, and duration, so nothing is re-derived here.
    """
    return HistoryEntry(
        title=(getattr(result, "title", "") or "").strip(),
        url=getattr(result, "url", "") or "",
        platform=getattr(result, "platform", "") or "",
        path=getattr(result, "path", "") or "",
        bytes=_as_int(getattr(result, "bytes", 0)),
        extension=(getattr(result, "extension", "") or "").lstrip("."),
        height=_as_int(getattr(result, "height", 0)),
        duration_seconds=_as_int(getattr(result, "duration_seconds", 0)),
        source=source,
    )


def status_of(entry, exists=None) -> str:
    """ok / missing / deleted. `exists` is injectable so this stays pure."""
    if entry.removed_at:
        return STATUS_DELETED
    check = exists if exists is not None else os.path.isfile
    return STATUS_OK if entry.path and check(entry.path) else STATUS_MISSING


def search(entries, text: str) -> List[HistoryEntry]:
    """Case-insensitive substring match over the title."""
    needle = (text or "").strip().lower()
    if not needle:
        return list(entries)
    return [e for e in entries if needle in (e.title or "").lower()]


def filter_entries(entries, platform: Optional[str] = None,
                   extension: Optional[str] = None) -> List[HistoryEntry]:
    out = list(entries)
    if platform:
        out = [e for e in out if e.platform == platform]
    if extension:
        want = extension.lower().lstrip(".")
        out = [e for e in out if (e.extension or "").lower().lstrip(".") == want]
    return out


_SORT_KEYS = {
    "title": lambda e, s: (e.title or "").lower(),
    "platform": lambda e, s: (e.platform or "").lower(),
    "extension": lambda e, s: (e.extension or "").lower(),
    "bytes": lambda e, s: _as_int(e.bytes),
    "created_at": lambda e, s: e.created_at or "",
    "status": lambda e, s: (s or {}).get(e.id, ""),
}


def sort_entries(entries, key: str, descending: bool = False,
                 statuses: Optional[dict] = None) -> List[HistoryEntry]:
    getter = _SORT_KEYS.get(key, _SORT_KEYS["created_at"])
    return sorted(entries, key=lambda e: getter(e, statuses), reverse=descending)


def _as_int(value) -> int:
    try:
        return int(value)
    except (TypeError, ValueError):
        return 0
```

# core/history.py  -- append

def _coerce_entries(raw) -> List[HistoryEntry]:
    """Rebuild entries, dropping unknown keys but never the entry."""
    if not isinstance(raw, list):
        return []
    known = HistoryEntry.__dataclass_fields__
    out: List[HistoryEntry] = []
    for item in raw:
        if not isinstance(item, dict):
            continue
        clean = {k: v for k, v in item.items() if k in known}
        try:
            entry = HistoryEntry(**clean)
        except (TypeError, ValueError):
            continue
        for name in ("bytes", "height", "duration_seconds"):
            setattr(entry, name, _as_int(getattr(entry, name)))
        out.append(entry)
    return out


class HistoryStore:
    """The history file. Atomic write, tolerant load, never drops an entry."""

    def __init__(self, path: Optional[str] = None):
        self.path = path if path is not None else history_path()
        self._entries: List[HistoryEntry] = []

    def load(self) -> None:
        self._entries = []
        if not self.path:
            return
        try:
            with open(self.path, "r", encoding="utf-8") as fh:
                data = json.load(fh)
        except (OSError, ValueError):
            return
        if not isinstance(data, dict) or data.get("version") != STORE_VERSION:
            return
        self._entries.extend(_coerce_entries(data.get("entries")))

    def save(self) -> None:
        if not self.path:
            return
        data = {"version": STORE_VERSION,
                "entries": [asdict(e) for e in self._entries]}
        tmp = f"{self.path}.{os.getpid()}.{uuid.uuid4().hex}.tmp"
        with open(tmp, "w", encoding="utf-8") as fh:
            json.dump(data, fh, indent=2)
        os.replace(tmp, self.path)

    def add(self, entry: HistoryEntry) -> HistoryEntry:
        self._entries.append(entry)
        self.save()
        return entry

    def remove(self, entry_id: str) -> bool:
        before = len(self._entries)
        self._entries = [e for e in self._entries if e.id != entry_id]
        if len(self._entries) == before:
            return False
        self.save()
        return True

    def mark_deleted(self, entry_id: str) -> bool:
        for entry in self._entries:
            if entry.id == entry_id:
                entry.removed_at = datetime.now(timezone.utc).isoformat()
                self.save()
                return True
        return False

    def entries(self) -> List[HistoryEntry]:
        return list(self._entries)


def record(store, result, source: str = "") -> Optional[HistoryEntry]:
    """Append a result to a store, swallowing any failure.

    Losing a history line must never lose a download, so this never raises.
    """
    if store is None:
        return None
    try:
        return store.add(entry_from_result(result, source))
    except Exception:
        return None
```

# utils/ui_helpers.py  -- append
def open_command(path: str, platform: str = None) -> list:
    """The command that opens `path` in the OS default application."""
    platform = platform or sys.platform
    if platform.startswith("win"):
        # `start` is a cmd builtin; the empty "" is the window title slot, or a
        # quoted path would be swallowed as the title.
        return ["cmd", "/c", "start", "", path]
    if platform == "darwin":
        return ["open", path]
    return ["xdg-open", path]


def open_file(path: str) -> bool:
    """Open a file with the OS default application. False on any failure."""
    if not path or not os.path.isfile(path):
        return False
    try:
        if sys.platform == "win32":
            os.startfile(path)          # noqa: S606  (Windows-only API)
        else:
            subprocess.Popen(open_command(path))
        return True
    except Exception:
        return False
```

from utils.ui_helpers import open_file, reveal_in_folder

_COLUMNS = ("Title", "Platform", "Format", "Size", "Date", "Status")
_STATUS_LABELS = {"ok": "OK", "missing": "Not found", "deleted": "Deleted by you"}


class HistoryTab(QtWidgets.QWidget):
    def __init__(self, store=None, parent=None):
        super().__init__(parent)
        self.store = store if store is not None else HistoryStore()
        self._entries = []
        self._statuses = {}
        self._sort_key = "created_at"
        self._sort_desc = True
        self._build()
        self.refresh()

    # -- construction ------------------------------------------------------
    def _build(self):
        layout = QtWidgets.QVBoxLayout(self)

        bar = QtWidgets.QHBoxLayout()
        self.search_box = QtWidgets.QLineEdit()
        self.search_box.setPlaceholderText("Search title...")
        self.search_box.setClearButtonEnabled(True)
        self.search_box.textChanged.connect(self.refresh)
        self.platform_box = QtWidgets.QComboBox()
        self.platform_box.currentIndexChanged.connect(self.refresh)
        self.format_box = QtWidgets.QComboBox()
        self.format_box.addItems(["All formats", "MP4", "MP3"])
        self.format_box.currentIndexChanged.connect(self.refresh)
        bar.addWidget(self.search_box, 1)
        bar.addWidget(self.platform_box)
        bar.addWidget(self.format_box)
        layout.addLayout(bar)

        self.table = QtWidgets.QTableWidget(0, len(_COLUMNS))
        self.table.setHorizontalHeaderLabels(_COLUMNS)
        self.table.setSelectionBehavior(QtWidgets.QAbstractItemView.SelectRows)
        self.table.setSelectionMode(QtWidgets.QAbstractItemView.SingleSelection)
        self.table.setEditTriggers(QtWidgets.QAbstractItemView.NoEditTriggers)
        self.table.verticalHeader().setVisible(False)
        self.table.horizontalHeader().setStretchLastSection(True)
        self.table.horizontalHeader().sectionClicked.connect(self._on_header)
        self.table.itemSelectionChanged.connect(self._sync_buttons)
        self.table.setContextMenuPolicy(QtCore.Qt.CustomContextMenu)
        self.table.customContextMenuRequested.connect(self._on_menu)
        layout.addWidget(self.table, 1)

        buttons = QtWidgets.QHBoxLayout()
        self.play_btn = QtWidgets.QPushButton("Play")
        self.play_btn.clicked.connect(self.play)
        self.folder_btn = QtWidgets.QPushButton("Open folder")
        self.folder_btn.clicked.connect(self.open_folder)
        self.details_btn = QtWidgets.QPushButton("Details")
        self.details_btn.clicked.connect(self.show_details)
        self.delete_btn = QtWidgets.QPushButton("Delete file")
        self.delete_btn.clicked.connect(self.delete_file)
        self.forget_btn = QtWidgets.QPushButton("Remove from history")
        self.forget_btn.clicked.connect(self.remove_entry)
        self.clear_btn = QtWidgets.QPushButton("Clear history")
        self.clear_btn.clicked.connect(self.clear_history)
        for button in (self.play_btn, self.folder_btn, self.details_btn):
            buttons.addWidget(button)
        buttons.addStretch(1)
        for button in (self.delete_btn, self.forget_btn, self.clear_btn):
            buttons.addWidget(button)
        layout.addLayout(buttons)

    # -- data --------------------------------------------------------------
    def refresh(self):
        self.store.load()
        entries = self.store.entries()
        self._statuses = {e.id: status_of(e) for e in entries}

        platforms = sorted({e.platform for e in entries if e.platform})
        self._sync_combo(self.platform_box, ["All platforms"] + platforms)

        fmt = self.format_box.currentText()
        entries = search(entries, self.search_box.text())
        entries = filter_entries(
            entries,
            platform=None if self.platform_box.currentIndex() <= 0
            else self.platform_box.currentText(),
            extension=None if fmt == "All formats" else fmt,
        )
        self._entries = sort_entries(entries, self._sort_key,
                                     self._sort_desc, self._statuses)
        self._fill_table()
        self._sync_buttons()

    def _sync_combo(self, combo, items):
        current = combo.currentText()
        if [combo.itemText(i) for i in range(combo.count())] == items:
            return
        combo.blockSignals(True)
        combo.clear()
        combo.addItems(items)
        index = combo.findText(current)
        combo.setCurrentIndex(index if index >= 0 else 0)
        combo.blockSignals(False)

    def _fill_table(self):
        self.table.setRowCount(len(self._entries))
        for row, entry in enumerate(self._entries):
            status = self._statuses.get(entry.id, "")
            cells = (
                entry.title or "(untitled)",
                entry.platform or "-",
                (entry.extension or "-").upper(),
                human_size(entry.bytes) if entry.bytes else "-",
                (entry.created_at or "")[:10],
                _STATUS_LABELS.get(status, status),
            )
            for col, text in enumerate(cells):
                item = QtWidgets.QTableWidgetItem(text)
                item.setData(QtCore.Qt.UserRole, entry.id)
                if status != STATUS_OK:
                    item.setForeground(QtGui.QBrush(QtGui.QColor("#8a8a8a")))
                self.table.setItem(row, col, item)
        self.table.resizeColumnsToContents()

    # -- selection ---------------------------------------------------------
    def _selected(self):
        rows = self.table.selectionModel().selectedRows()
        if not rows:
            return None
        entry_id = self.table.item(rows[0].row(), 0).data(QtCore.Qt.UserRole)
        for entry in self._entries:
            if entry.id == entry_id:
                return entry
        return None

    def _sync_buttons(self):
        entry = self._selected()
        usable = bool(entry) and self._statuses.get(entry.id) == STATUS_OK
        for button in (self.play_btn, self.folder_btn, self.delete_btn):
            button.setEnabled(usable)
        self.details_btn.setEnabled(bool(entry))
        self.forget_btn.setEnabled(bool(entry))
        self.clear_btn.setEnabled(bool(self._entries))

    def _on_header(self, index):
        key = ("title", "platform", "extension", "bytes", "created_at",
               "status")[index]
        if key == self._sort_key:
            self._sort_desc = not self._sort_desc
        else:
            self._sort_key, self._sort_desc = key, False
        self.refresh()

    # -- actions -----------------------------------------------------------
    def play(self):
        entry = self._selected()
        if entry and not open_file(entry.path):
            QtWidgets.QMessageBox.warning(
                self, "Cannot play",
                "No application is set to open this file type.",
            )

    def open_folder(self):
        entry = self._selected()
        if entry:
            reveal_in_folder(entry.path)

    def delete_file(self):
        entry = self._selected()
        if not entry or self._statuses.get(entry.id) != STATUS_OK:
            return
        name = os.path.basename(entry.path)
        answer = QtWidgets.QMessageBox.question(
            self, "Delete file",
            f"Delete {name} from disk?\n\nThe history entry is kept.",
            QtWidgets.QMessageBox.Yes | QtWidgets.QMessageBox.No,
        )
        if answer != QtWidgets.QMessageBox.Yes:
            return
        try:
            os.remove(entry.path)
        except OSError as exc:
            QtWidgets.QMessageBox.warning(self, "Delete failed", str(exc))
            return
        self.store.mark_deleted(entry.id)
        self.refresh()

    def remove_entry(self):
        entry = self._selected()
        if not entry:
            return
        answer = QtWidgets.QMessageBox.question(
            self, "Remove from history",
            "Remove this entry from the history?\n\nThe file is not deleted.",
            QtWidgets.QMessageBox.Yes | QtWidgets.QMessageBox.No,
        )
        if answer != QtWidgets.QMessageBox.Yes:
            return
        self.store.remove(entry.id)
        self.refresh()

    def clear_history(self):
        if not self._entries:
            return
        answer = QtWidgets.QMessageBox.question(
            self, "Clear history",
            "Remove every entry from the history?\n\nNo files are deleted.",
            QtWidgets.QMessageBox.Yes | QtWidgets.QMessageBox.No,
        )
        if answer != QtWidgets.QMessageBox.Yes:
            return
        for entry in self.store.entries():
            self.store.remove(entry.id)
        self.refresh()

    def show_details(self):
        entry = self._selected()
        if not entry:
            return
        lines = [
            ("Title", entry.title or "(untitled)"),
            ("URL", entry.url or "-"),
            ("Platform", entry.platform or "-"),
            ("File", entry.path or "-"),
            ("Format", (entry.extension or "-").upper()),
            ("Resolution", f"{entry.height}p" if entry.height else "-"),
            ("Duration", _duration(entry.duration_seconds)),
            ("Size", human_size(entry.bytes) if entry.bytes else "-"),
            ("Downloaded", entry.created_at or "-"),
            ("Source", entry.source or "-"),
            ("Status", _STATUS_LABELS.get(self._statuses.get(entry.id, ""), "-")),
        ]
        if self._statuses.get(entry.id) == STATUS_OK:
            probed = probe_media(entry.path) or {}
            lines += [
                ("Container", probed.get("container") or "-"),
                ("Video codec", probed.get("vcodec") or "-"),
                ("Audio codec", probed.get("acodec") or "-"),
            ]
        text = "\n".join(f"{name}: {value}" for name, value in lines)
        QtWidgets.QMessageBox.information(self, "Download details", text)

    def _on_menu(self, pos):
        if self._selected() is None:
            return
        menu = QtWidgets.QMenu(self)
        for button in (self.play_btn, self.folder_btn, self.details_btn,
                       self.delete_btn, self.forget_btn):
            action = menu.addAction(button.text())
            action.setEnabled(button.isEnabled())
            action.triggered.connect(button.click)
        menu.exec_(self.table.viewport().mapToGlobal(pos))


def _duration(seconds: int) -> str:
    if not seconds:
        return "-"
    minutes, secs = divmod(int(seconds), 60)
    hours, minutes = divmod(minutes, 60)
    if hours:
        return f"{hours}:{minutes:02d}:{secs:02d}"
    return f"{minutes}:{secs:02d}"
```
