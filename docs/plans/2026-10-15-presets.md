# Download Presets Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Save the whole download setup as a named preset, apply it in one click from the Download and Queue tabs, and use it from the CLI, with editor-first built-ins.

**Architecture:** A Qt-free `core/presets.py` holds the `Preset` model, the five built-ins, a versioned `PresetStore` over `presets.json`, and the app-directory path helper. The Download tab gains a Preset row (select, save, manage) and the Queue tab a selector; both apply a preset into their existing controls. The CLI gains `--preset`.

**Tech Stack:** Python 3.8+, PyQt5, stdlib `json`, pytest. No new dependencies.

**Spec:** `docs/specs/2026-10-15-presets-design.md`

## Global Constraints

- `core/` must not import PyQt5. Enforced by `tests/test_core_isolation.py`.
- No new runtime dependencies.
- Storage is `presets.json` in the app directory (`default_cookie_dir()`, next to `queue.json`); the last-used preset name lives in `QSettings` under `preset/last`.
- Built-ins live in code and cannot be shadowed, edited, or deleted; the file holds user presets only.
- CLI precedence is explicit flag > preset > default.
- The download result contract stays `schema_version` `"1.0"`.
- All existing tests stay green (`python -m pytest -q`); flake8 `--select=E9,F63,F7,F82` prints `0`.
- Conventional Commits. Never push without asking.

```python
"""Named download presets: editor built-ins plus user presets in a JSON file.

Pure model and store; no Qt. The UI and the CLI apply a preset to their controls.
"""

import json
import os
from dataclasses import asdict, dataclass
from typing import List, Optional

STORE_VERSION = 1


@dataclass
class Preset:
    name: str
    output: str = "MP4"
    resolution: str = "1080"
    convert: bool = True
    add_channel: bool = False
    add_timestamp: bool = False
    save_dir: str = ""
    cookies: str = ""
    builtin: bool = False


BUILTIN_PRESETS = [
    Preset(name="Editor-safe 1080p", output="MP4", resolution="1080",
           convert=True, builtin=True),
    Preset(name="Editor-safe 720p", output="MP4", resolution="720",
           convert=True, builtin=True),
    Preset(name="Audio only (MP3)", output="MP3", resolution="1080",
           convert=False, builtin=True),
    Preset(name="Original quality", output="MP4", resolution="1080",
           convert=False, builtin=True),
    Preset(name="Archive master", output="MP4", resolution="1080",
           convert=False, add_channel=True, add_timestamp=True, builtin=True),
]

_BUILTIN_NAMES = {p.name for p in BUILTIN_PRESETS}


def default_presets_path() -> str:
    from sites.cookies import default_cookie_dir
    return os.path.join(default_cookie_dir(), "presets.json")


def preset_to_dict(preset: Preset) -> dict:
    data = asdict(preset)
    data.pop("builtin", None)
    return data


def preset_from_dict(data) -> Optional[Preset]:
    data = data or {}
    name = data.get("name") or ""
    if not name:
        return None
    return Preset(
        name=name,
        output=data.get("output") or "MP4",
        resolution=data.get("resolution") or "1080",
        convert=bool(data.get("convert", True)),
        add_channel=bool(data.get("add_channel", False)),
        add_timestamp=bool(data.get("add_timestamp", False)),
        save_dir=data.get("save_dir") or "",
        cookies=data.get("cookies") or "",
        builtin=False,
    )


class PresetStore:
    def __init__(self, path: str):
        self.path = path
        self.presets: List[Preset] = []

    def load(self) -> None:
        self.presets = []
        if not self.path:
            return
        try:
            with open(self.path, "r", encoding="utf-8") as fh:
                data = json.load(fh)
        except (OSError, ValueError):
            return
        if not isinstance(data, dict) or data.get("version") != STORE_VERSION:
            return
        for raw in data.get("presets") or []:
            preset = preset_from_dict(raw)
            if preset and preset.name not in _BUILTIN_NAMES:
                self.presets.append(preset)

    def save(self) -> None:
        if not self.path:
            return
        data = {
            "version": STORE_VERSION,
            "presets": [preset_to_dict(p) for p in self.presets],
        }
        tmp = "%s.%d.tmp" % (self.path, os.getpid())
        try:
            with open(tmp, "w", encoding="utf-8") as fh:
                json.dump(data, fh, indent=2)
            os.replace(tmp, self.path)
        except OSError:
            pass

    def user(self) -> List[Preset]:
        return list(self.presets)

    def all(self) -> List[Preset]:
        return list(BUILTIN_PRESETS) + list(self.presets)

    def get(self, name: str) -> Optional[Preset]:
        for preset in self.all():
            if preset.name == name:
                return preset
        return None

    def add(self, preset: Preset) -> bool:
        if preset.name in _BUILTIN_NAMES:
            return False
        for index, existing in enumerate(self.presets):
            if existing.name == preset.name:
                self.presets[index] = preset
                return True
        self.presets.append(preset)
        return True

    def remove(self, name: str) -> bool:
        for index, preset in enumerate(self.presets):
            if preset.name == name:
                del self.presets[index]
                return True
        return False

    def rename(self, old: str, new: str) -> bool:
        if not new or new in _BUILTIN_NAMES:
            return False
        for preset in self.presets:
            if preset.name == old:
                preset.name = new
                return True
        return False
```

def manage_presets(parent, store) -> bool:
    """Rename or delete user presets. Returns True if anything changed."""
    changed = False
    dialog = QtWidgets.QDialog(parent)
    dialog.setWindowTitle("Manage presets")
    dialog.resize(420, 320)
    layout = QtWidgets.QVBoxLayout(dialog)
    layout.addWidget(QtWidgets.QLabel("Your presets:"))
    listing = QtWidgets.QListWidget()
    for preset in store.user():
        listing.addItem(preset.name)
    layout.addWidget(listing)

    buttons = QtWidgets.QHBoxLayout()
    rename_btn = QtWidgets.QPushButton("Rename")
    delete_btn = QtWidgets.QPushButton("Delete")
    buttons.addWidget(rename_btn)
    buttons.addWidget(delete_btn)
    buttons.addStretch(1)
    close_btn = QtWidgets.QPushButton("Close")
    buttons.addWidget(close_btn)
    layout.addLayout(buttons)

    def selected_name():
        item = listing.currentItem()
        return item.text() if item else ""

    def on_rename():
        nonlocal changed
        old = selected_name()
        if not old:
            return
        new, ok = QtWidgets.QInputDialog.getText(
            dialog, "Rename preset", "New name:", text=old
        )
        if ok and new.strip() and store.rename(old, new.strip()):
            changed = True
            listing.clear()
            for preset in store.user():
                listing.addItem(preset.name)

    def on_delete():
        nonlocal changed
        name = selected_name()
        if not name:
            return
        confirm = QtWidgets.QMessageBox.question(
            dialog, "Delete preset", "Delete '%s'?" % name
        )
        if confirm == QtWidgets.QMessageBox.Yes and store.remove(name):
            changed = True
            listing.clear()
            for preset in store.user():
                listing.addItem(preset.name)

    rename_btn.clicked.connect(on_rename)
    delete_btn.clicked.connect(on_delete)
    close_btn.clicked.connect(dialog.accept)
    dialog.exec_()
    return changed
```

    def _current_preset_name(self):
        return self.settings.value("preset/last", "")

    def _refresh_presets(self):
        self._applying_preset = True
        self.preset_combo.clear()
        for preset in self.preset_store.all():
            self.preset_combo.addItem(preset.name, preset.name)
        self.preset_combo.addItem("Custom", None)
        last = self._current_preset_name()
        index = self.preset_combo.findData(last) if last else -1
        self.preset_combo.setCurrentIndex(index if index >= 0 else self.preset_combo.count() - 1)
        self._applying_preset = False

    def on_preset_selected(self, _index):
        if self._applying_preset:
            return
        name = self.preset_combo.currentData()
        if not name:
            return
        preset = self.preset_store.get(name)
        if preset:
            self._apply_preset(preset)

    def _apply_preset(self, preset):
        self._applying_preset = True
        self.output_combo.setCurrentText(preset.output)
        self.resolution_combo.setCurrentText(preset.resolution)
        self.convert_checkbox.setChecked(preset.convert)
        self.checkbox_channel.setChecked(preset.add_channel)
        self.checkbox_timestamp.setChecked(preset.add_timestamp)
        self.save_dir_input.setText(preset.save_dir or default_download_folder())
        self.cookies_input.setText(preset.cookies or "")
        self.cookies_file_ref["path"] = preset.cookies or ""
        self._applying_preset = False
        self.settings.setValue("preset/last", preset.name)

    def _mark_custom(self, *_args):
        if self._applying_preset:
            return
        index = self.preset_combo.findData(None)
        if index >= 0 and self.preset_combo.currentIndex() != index:
            self._applying_preset = True
            self.preset_combo.setCurrentIndex(index)
            self._applying_preset = False

    def _current_settings_preset(self, name):
        return Preset(
            name=name,
            output=self.output_combo.currentText(),
            resolution=self.resolution_combo.currentText(),
            convert=self.convert_checkbox.isChecked(),
            add_channel=self.checkbox_channel.isChecked(),
            add_timestamp=self.checkbox_timestamp.isChecked(),
            save_dir=self.save_dir_input.text().strip(),
            cookies=self.cookies_input.text().strip(),
        )

    def on_save_preset(self):
        name, ok = QtWidgets.QInputDialog.getText(
            self, "Save preset", "Preset name:"
        )
        name = (name or "").strip()
        if not ok or not name:
            return
        if self.preset_store.get(name) is not None and name in [
            p.name for p in self.preset_store.user()
        ]:
            confirm = QtWidgets.QMessageBox.question(
                self, "Overwrite preset", "Overwrite '%s'?" % name
            )
            if confirm != QtWidgets.QMessageBox.Yes:
                return
        if not self.preset_store.add(self._current_settings_preset(name)):
            self.show_hint("That name is a built-in preset; pick another.")
            return
        self.preset_store.save()
        self.settings.setValue("preset/last", name)
        self._refresh_presets()
        self.preset_combo.setCurrentIndex(self.preset_combo.findData(name))

    def on_manage_presets(self):
        if manage_presets(self, self.preset_store):
            self.preset_store.save()
            self._refresh_presets()
```
