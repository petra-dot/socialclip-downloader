def _table() -> dict:
    return _DARK if is_dark() else _LIGHT


def token(name: str) -> str:
    """The current theme's value for a semantic token name."""
    return _table().get(name, "")


def space(name: str) -> int:
    return SPACING.get(name, SPACING["md"])


def state_color(kind: str) -> str:
    """A hex color for a UI state. Now a thin alias of `token`."""
    return token(kind)


def _relative_luminance(hex_color: str) -> float:
    channels = [int(hex_color[i:i + 2], 16) / 255.0 for i in (1, 3, 5)]

    def linear(c):
        return c / 12.92 if c <= 0.03928 else ((c + 0.055) / 1.055) ** 2.4

    r, g, b = (linear(c) for c in channels)
    return 0.2126 * r + 0.7152 * g + 0.0722 * b


def contrast_ratio(fg: str, bg: str) -> float:
    """WCAG contrast ratio between two #rrggbb colors. Pure."""
    a, b = _relative_luminance(fg), _relative_luminance(bg)
    hi, lo = max(a, b), min(a, b)
    return (hi + 0.05) / (lo + 0.05)


def make_chip(text: str, tone: str = "neutral"):
    """A rounded pill; the child QLabel is exposed as `.label`."""
    color = token(tone) if tone != "neutral" else token("text_secondary")
    border = token(tone) if tone != "neutral" else token("border")
    label = QtWidgets.QLabel(text)
    label.setStyleSheet(
        "border: 1px solid %s; border-radius: %dpx; padding: 1px 8px; color: %s;"
        % (border, RADIUS["chip"], color)
    )
    label.setAlignment(QtCore.Qt.AlignCenter)
    holder = QtWidgets.QWidget()
    row = QtWidgets.QHBoxLayout(holder)
    row.setContentsMargins(4, 0, 4, 0)
    row.addStretch(1)
    row.addWidget(label)
    row.addStretch(1)
    holder.label = label
    return holder


def make_result_panel():
    """The right-hand result panel shared by the Download and Convert tabs.

    Returns (frame, labels) where labels has keys `caption`, `body`, `ready`.
    The caller owns the progress bar, the action buttons, and the state logic.
    """
    frame = QtWidgets.QFrame()
    frame.setFrameShape(QtWidgets.QFrame.StyledPanel)
    frame.setBackgroundRole(QtGui.QPalette.AlternateBase)
    frame.setAutoFillBackground(True)
    frame.setMinimumWidth(240)
    box = QtWidgets.QVBoxLayout(frame)
    box.setContentsMargins(space("lg"), space("md"), space("lg"), space("md"))
    box.setSpacing(space("sm"))

    caption = QtWidgets.QLabel("Result")
    caption.setStyleSheet("color: %s;" % token("text_muted"))
    box.addWidget(caption)
    body = QtWidgets.QLabel("")
    body.setWordWrap(True)
    box.addWidget(body)
    ready = QtWidgets.QLabel("")
    ready.setWordWrap(True)
    ready.setVisible(False)
    box.addWidget(ready)
    box.addStretch(1)
    return frame, {"caption": caption, "body": body, "ready": ready,
                   "layout": box}


def primary_button_style() -> str:
    """Accent-filled primary button, theme-aware."""
    a, h, p = token("accent"), token("accent_hover"), token("accent_pressed")
    on, border, muted = token("on_accent"), token("border"), token("text_muted")
    r = RADIUS["control"]
    return (
        "QPushButton { background: %(a)s; color: %(on)s; border: 1px solid %(a)s;"
        " border-radius: %(r)dpx; padding: 5px 14px; font-weight: 600; }"
        "QPushButton:hover { background: %(h)s; border-color: %(h)s; }"
        "QPushButton:pressed { background: %(p)s; border-color: %(p)s; }"
        "QPushButton:disabled { background: %(b)s; border-color: %(b)s;"
        " color: %(m)s; }"

def make_title(label):
    font = label.font()
    font.setBold(True)
    base = QtWidgets.QApplication.font().pointSizeF()
    if base and base > 0:
        font.setPointSizeF(base + 2.0)
    label.setFont(font)
    return label


def make_caption(label):
    label.setStyleSheet("color: %s;" % token("text_muted"))
    return label

# utils/system_theme.py
"""Detect whether the OS prefers a dark or light interface.

Qt 5 has no portable API for this, so each platform is read directly. Never
raises: an unknown or unreadable answer is "light". Qt-free and injectable.
"""

import subprocess
import sys

from utils.proc import no_window_kwargs


def _default_runner(cmd: str) -> str:
    result = subprocess.run(
        cmd, shell=True, stdout=subprocess.PIPE, stderr=subprocess.DEVNULL,
        text=True, **no_window_kwargs(),
    )
    return result.stdout or ""


def _windows_is_dark() -> str:
    try:
        import winreg
        key = winreg.OpenKey(
            winreg.HKEY_CURRENT_USER,
            r"Software\Microsoft\Windows\CurrentVersion\Themes\Personalize",
        )
        try:
            value, _ = winreg.QueryValueEx(key, "AppsUseLightTheme")
        finally:
            winreg.CloseKey(key)
        return "light" if int(value) else "dark"
    except Exception:
        return ""


def _windows_answer(run) -> str:
    if run is not _default_runner:
        # Tests inject a runner; treat its output like the registry value.
        raw = (run("AppsUseLightTheme") or "").strip()
        return "light" if raw == "1" else ("dark" if raw == "0" else "")
    return _windows_is_dark()


def detect(runner=None, platform=None) -> str:
    """'dark' or 'light' for the OS preference."""
    platform = sys.platform if platform is None else platform
    run = _default_runner if runner is None else runner
    try:
        if platform.startswith("win"):
            answer = _windows_answer(run)
        elif platform == "darwin":
            answer = "dark" if "dark" in run(
                "defaults read -g AppleInterfaceStyle").lower() else "light"
        elif platform.startswith("linux"):
            answer = "dark" if "dark" in run(
                "gsettings get org.gnome.desktop.interface color-scheme").lower() \
                else "light"
        else:
            answer = ""
    except Exception:
        answer = ""
    return answer if answer in ("dark", "light") else "light"
```

The injected-runner win32 branch is part of the implementation, so the tests
pass at the first run - there is no separate fix step.

    p.setColor(QtGui.QPalette.Window, QtGui.QColor(colors["window"]))
    p.setColor(QtGui.QPalette.WindowText, QtGui.QColor(colors["text"]))

    p.setColor(QtGui.QPalette.Button, QtGui.QColor(colors["window"]))
    p.setColor(QtGui.QPalette.ButtonText, QtGui.QColor(colors["text"]))
    p.setColor(QtGui.QPalette.BrightText, QtGui.QColor(colors["on_accent"]))
    p.setColor(QtGui.QPalette.ToolTipBase, QtGui.QColor(colors["base"]))
    p.setColor(QtGui.QPalette.ToolTipText, QtGui.QColor(colors["text"]))
    p.setColor(QtGui.QPalette.Highlight, QtGui.QColor(colors["accent"]))
    p.setColor(QtGui.QPalette.HighlightedText, QtGui.QColor(colors["on_accent"]))
    p.setColor(QtGui.QPalette.PlaceholderText, QtGui.QColor(colors["text_muted"]))
    for role in (QtGui.QPalette.Text, QtGui.QPalette.ButtonText,
                 QtGui.QPalette.WindowText):
        p.setColor(QtGui.QPalette.Disabled, role,

    p.setColor(QtGui.QPalette.Disabled, QtGui.QPalette.Base,
               QtGui.QColor(colors["window"]))

def tab_bar_style() -> str:
    """The active tab carries the accent, so it recolors with the theme."""
    return (
        "QTabBar::tab:selected { color: %(a)s; border-bottom: 2px solid %(a)s; }"
        "QTabBar::tab { border-bottom: 2px solid transparent; padding: 5px 12px; }"
        % {"a": token("accent")}
    )


def apply_theme(app, mode: str):
    """Apply a theme. Fusion for every mode so widget shapes never change."""
    global _current_mode
    resolved = mode
    if mode == THEME_SYSTEM:
        from utils.system_theme import detect
        resolved = detect()
    _current_mode = THEME_DARK if resolved == THEME_DARK else THEME_LIGHT
    app.setStyle("Fusion")
    app.setPalette(_palette(_DARK if resolved == THEME_DARK else _LIGHT))
    app.setStyleSheet(tab_bar_style())
```
